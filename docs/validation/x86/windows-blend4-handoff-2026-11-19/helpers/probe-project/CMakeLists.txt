cmake_minimum_required(VERSION 3.20)
project(FrozenMovementRegression LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Execute this supplemental check on native x64 Windows")
endif()
if(NOT DEFINED VCL_CHECKOUT)
  message(FATAL_ERROR "Set VCL_CHECKOUT to the verified frozen source root")
endif()
enable_testing()
set(VCL_BACKEND 1 CACHE STRING "" FORCE)
set(VCL_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(VCL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(VCL_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(VCL_TESTBENCH_DIR "" CACHE PATH "" FORCE)
# Reuse the frozen suite's actual options; do not duplicate or relax its FP flags.
add_subdirectory("${VCL_CHECKOUT}" frozen-library)
foreach(route facade originals)
  foreach(scope default namespace)
    set(target probe_movement_${route}_${scope})
    add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/movement-${route}.cpp")
    target_link_libraries(${target} PRIVATE vcl_test_options)
    if(scope STREQUAL "namespace")
      target_compile_definitions(${target} PRIVATE VCL_NAMESPACE=isolated)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 60)
  endforeach()
endforeach()
